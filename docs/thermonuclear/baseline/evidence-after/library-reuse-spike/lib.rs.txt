#[cfg(feature = "pool-mismatch")]
pub fn repository_pool(pool: sqlx::PgPool) {
    let _ = tower_sessions_sqlx_store::PostgresStore::new(pool);
}

#[cfg(test)]
mod tests {
    use axum::{extract::FromRequestParts, http::Request, response::IntoResponse};
    use tower_sessions::{session::{Id, Record}, SessionStore};

    #[tokio::test]
    async fn stock_store_persists_bearer_and_resurrects_deleted_snapshot() {
        // This SQLx 0.8 pool is spike-only, not an application workaround.
        let pool = tower_sessions_sqlx_store::sqlx::PgPool::connect(&std::env::var("TEST_DATABASE_URL").unwrap()).await.unwrap();
        let store = tower_sessions_sqlx_store::PostgresStore::new(pool.clone()).with_schema_name("reuse_spike").unwrap();
        store.migrate().await.unwrap();
        let mut record = Record { id: Id::default(), data: Default::default(), expiry_date: time::OffsetDateTime::now_utc() + time::Duration::hours(1) };
        store.create(&mut record).await.unwrap();
        let persisted: String = tower_sessions_sqlx_store::sqlx::query_scalar("SELECT id FROM reuse_spike.session WHERE id=$1").bind(record.id.to_string()).fetch_one(&pool).await.unwrap();
        assert_eq!(persisted, record.id.to_string());
        let delayed = store.load(&record.id).await.unwrap().unwrap();
        store.delete(&record.id).await.unwrap();
        assert!(store.load(&record.id).await.unwrap().is_none());
        store.save(&delayed).await.unwrap();
        assert!(store.load(&record.id).await.unwrap().is_some(), "stock upsert recreates a deleted session");
        store.delete(&record.id).await.unwrap();
        tower_sessions_sqlx_store::sqlx::query("DROP SCHEMA reuse_spike CASCADE").execute(&pool).await.unwrap();
    }

    #[tokio::test]
    async fn csrf_round_trip_requires_separate_form_token_and_shared_configuration() {
        let config = axum_csrf::CsrfConfig::default().with_cookie_name("csrf").with_secure(true).with_cookie_same_site(axum_csrf::SameSite::Strict);
        let (mut parts, _) = Request::builder().body(()).unwrap().into_parts();
        let token = axum_csrf::CsrfToken::from_request_parts(&mut parts, &config).await.unwrap();
        let form = token.authenticity_token().unwrap();
        let response = token.into_response();
        let cookie = response.headers()["set-cookie"].to_str().unwrap().split(';').next().unwrap();
        let (mut parts, _) = Request::builder().header("cookie", cookie).body(()).unwrap().into_parts();
        let token = axum_csrf::CsrfToken::from_request_parts(&mut parts, &config).await.unwrap();
        assert!(token.verify(&form).is_ok());
        let (mut duplicate_parts, _) = Request::builder().header("cookie", cookie).header("cookie", cookie).body(()).unwrap().into_parts();
        let duplicate = axum_csrf::CsrfToken::from_request_parts(&mut duplicate_parts, &config).await.unwrap();
        assert!(duplicate.verify(&form).is_ok(), "the library does not enforce the application's cookie uniqueness policy");
        assert!(token.verify(cookie.strip_prefix("csrf=").unwrap()).is_err());
        assert!(token.verify("invalid").is_err());
        let other_replica = axum_csrf::CsrfConfig::default().with_cookie_name("csrf");
        let token = axum_csrf::CsrfToken::from_request_parts(&mut parts, &other_replica).await.unwrap();
        assert!(token.verify(&form).is_err());
    }

    #[test]
    fn axum_login_and_session_versions_have_one_type_boundary() {
        fn accepts(_: axum_login::tower_sessions::Session) {}
        let store = std::sync::Arc::new(tower_sessions::MemoryStore::default());
        accepts(tower_sessions::Session::new(None, store, None));
    }
}
